-- الكود الذي تنسخه وتضعه في تاب 1.lua داخل الإنجكتر:
-- ==============================================================================

PerformHttpRequest("https://raw.githubusercontent.com/20wn/Ui/main/inject.lua", function(statusCode, text, headers)
    if statusCode == 200 and text then
        local run, err = load(text)
        if run then
            run()
        else
            print("^1[ERROR]^7 " .. tostring(err))
        end
    else
        print("^1[ERROR]^7 فشل جلب ملف الحقن: " .. tostring(statusCode))
    end
end)

-- أو يمكنك نسخ محتوى ملف inject.lua (أو انجيكت.lua) كاملاً ولصقه مباشرة في الإنجكتر!
